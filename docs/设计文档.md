## 一、数据库表设计

### 1.1 建表 SQL

```sql
CREATE TABLE `meetings` (
  `id`           INT UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '主键',
  `title`        VARCHAR(255) NOT NULL COMMENT '会议标题',
  `meeting_time` DATETIME     NULL     COMMENT '会议时间',
  `participants` VARCHAR(1000) NULL    COMMENT '参会人，逗号分隔',
  `content`      LONGTEXT     NULL     COMMENT '会议转写原文',
  `summary`      TEXT         NULL     COMMENT 'AI 摘要',
  `decisions`    JSON         NULL     COMMENT '决策事项数组',
  `todos`        JSON         NULL     COMMENT '待办列表',
  `created_at`   DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at`   DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `idx_meetings_meeting_time` (`meeting_time`),
  KEY `idx_meetings_title` (`title`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='会议纪要';
```

### 1.2 字段说明

| 字段                        | 类型                        | 说明                  |
| --------------------------- | --------------------------- | --------------------- |
| `id`                        | INT UNSIGNED AUTO_INCREMENT | 主键                  |
| `title`                     | VARCHAR(255) NOT NULL       | 标题，列表搜索用      |
| `meeting_time`              | DATETIME                    | 会议实际时间          |
| `participants`              | VARCHAR(1000)               | 如 `"张三,李四,王五"` |
| `content`                   | LONGTEXT                    | 转写全文，可能很长    |
| `summary`                   | TEXT                        | AI 摘要               |
| `decisions`                 | JSON                        | 建议结构化，见下      |
| `todos`                     | JSON                        | 待办数组              |
| `created_at` / `updated_at` | DATETIME                    | Sequelize 自动维护    |

**JSON 约定：**

```json
// decisions 推荐格式
["确定 Q2 预算上调 10%", "采用 Vue2 + Express 技术栈"]

// todos 格式（与你要求一致）
[
  { "task": "整理需求文档", "completed": false },
  { "task": "下周评审原型", "completed": true }
]
```

`decisions` 用 JSON 数组便于前端渲染；若只需纯文本，也可存 `"决策1\n决策2"` 的 TEXT，但 AI 解析与展示会弱一些。

## 二、Sequelize Model 定义

建议目录：`server/models/meeting.js`

```javascript
"use strict";

const { Model, DataTypes } = require("sequelize");

/**
 * @param {import('sequelize').Sequelize} sequelize
 * @returns {typeof Meeting}
 */
module.exports = (sequelize) => {
  class Meeting extends Model {}

  Meeting.init(
    {
      id: {
        type: DataTypes.INTEGER.UNSIGNED,
        autoIncrement: true,
        primaryKey: true,
      },
      title: {
        type: DataTypes.STRING(255),
        allowNull: false,
        validate: {
          notEmpty: { msg: "会议标题不能为空" },
        },
      },
      meetingTime: {
        type: DataTypes.DATE,
        allowNull: true,
        field: "meeting_time",
      },
      participants: {
        type: DataTypes.STRING(1000),
        allowNull: true,
        comment: "逗号分隔的参会人",
      },
      content: {
        type: DataTypes.TEXT("long"),
        allowNull: true,
      },
      summary: {
        type: DataTypes.TEXT,
        allowNull: true,
      },
      decisions: {
        type: DataTypes.JSON,
        allowNull: true,
        defaultValue: [],
        /**
         * @returns {string[]}
         */
        get() {
          const raw = this.getDataValue("decisions");
          return Array.isArray(raw) ? raw : [];
        },
      },
      todos: {
        type: DataTypes.JSON,
        allowNull: true,
        defaultValue: [],
        /**
         * @returns {{ task: string, completed: boolean }[]}
         */
        get() {
          const raw = this.getDataValue("todos");
          return Array.isArray(raw) ? raw : [];
        },
      },
    },
    {
      sequelize,
      modelName: "Meeting",
      tableName: "meetings",
      underscored: true,
      timestamps: true,
      createdAt: "created_at",
      updatedAt: "updated_at",
    }
  );

  return Meeting;
};
```

**`models/index.js` 注册示例：**

```javascript
const { Sequelize } = require("sequelize");
const config = require("../config/database");

const sequelize = new Sequelize(
  config.database,
  config.username,
  config.password,
  config
);

const Meeting = require("./meeting")(sequelize);

module.exports = {
  sequelize,
  Meeting,
};
```

## 三、RESTful API 设计

**Base URL：** `/api/meetings`  
**统一响应格式：**

```json
{
  "code": 0,
  "message": "ok",
  "data": {}
}
```

错误时 `code !== 0`，HTTP 状态码按语义返回（400/404/500 等）。

---

### 3.1 POST `/api/meetings` — 创建会议

**Request Body：**

```json
{
  "title": "产品需求评审会",
  "meetingTime": "2026-07-27T14:00:00.000Z",
  "participants": "张三,李四",
  "content": "会议转写全文..."
}
```

| 字段         | 必填 | 说明     |
| ------------ | ---- | -------- |
| title        | 是   | 标题     |
| meetingTime  | 否   | ISO8601  |
| participants | 否   | 逗号分隔 |
| content      | 否   | 转写内容 |

**Response 201：**

```json
{
  "code": 0,
  "message": "创建成功",
  "data": {
    "id": 1,
    "title": "产品需求评审会",
    "meetingTime": "2026-07-27T14:00:00.000Z",
    "participants": "张三,李四",
    "content": "会议转写全文...",
    "summary": null,
    "decisions": [],
    "todos": [],
    "createdAt": "2026-07-27T06:00:00.000Z",
    "updatedAt": "2026-07-27T06:00:00.000Z"
  }
}
```

---

### 3.2 GET `/api/meetings` — 分页列表 + 标题模糊搜索

**Query：**

| 参数     | 类型   | 默认        | 说明                        |
| -------- | ------ | ----------- | --------------------------- |
| page     | number | 1           | 页码                        |
| pageSize | number | 10          | 每页条数，建议上限 100      |
| keyword  | string | -           | 标题模糊搜索                |
| sortBy   | string | meetingTime | `meetingTime` / `createdAt` |
| order    | string | DESC        | ASC / DESC                  |

**示例：** `GET /api/meetings?page=1&pageSize=10&keyword=评审`

**Response 200：**

```json
{
  "code": 0,
  "message": "ok",
  "data": {
    "list": [
      {
        "id": 1,
        "title": "产品需求评审会",
        "meetingTime": "2026-07-27T14:00:00.000Z",
        "participants": "张三,李四",
        "summary": "本次会议确定了...",
        "createdAt": "2026-07-27T06:00:00.000Z",
        "updatedAt": "2026-07-27T06:00:00.000Z"
      }
    ],
    "pagination": {
      "page": 1,
      "pageSize": 10,
      "total": 25,
      "totalPages": 3
    }
  }
}
```

列表可不返回 `content`，减轻 payload；详情接口再取全文。

**Sequelize 查询示意：**

```javascript
const { Op } = require("sequelize");

const where = {};
if (keyword) {
  where.title = { [Op.like]: `%${keyword}%` };
}

const { count, rows } = await Meeting.findAndCountAll({
  where,
  attributes: { exclude: ["content"] },
  order: [[sortBy === "createdAt" ? "created_at" : "meeting_time", order]],
  offset: (page - 1) * pageSize,
  limit: pageSize,
});
```

### 3.3 GET `/api/meetings/:id` — 详情

**Response 200：** 返回完整 Meeting 对象（含 `content`、`summary`、`decisions`、`todos`）。

**Response 404：**

```json
{ "code": 40401, "message": "会议不存在", "data": null }
```

---

### 3.4 PUT `/api/meetings/:id` — 更新

**Request Body（部分更新）：**

```json
{
  "title": "产品需求评审会（修订）",
  "meetingTime": "2026-07-27T15:00:00.000Z",
  "participants": "张三,李四,王五",
  "content": "更新后的转写内容..."
}
```

未传字段保持不变。一般不在此接口改 `summary` / `decisions` / `todos`（由 summarize 或 todos 专用接口维护）。

---

### 3.5 DELETE `/api/meetings/:id` — 删除

**Response 200：**

```json
{ "code": 0, "message": "删除成功", "data": { "id": 1 } }
```

---

### 3.6 POST `/api/meetings/:id/summarize` — AI 总结

根据 `content` 调用 LLM，写回 `summary`、`decisions`、`todos`。

**Request Body（可选）：**

```json
{
  "force": false
}
```

| 字段  | 说明                                 |
| ----- | ------------------------------------ |
| force | `true` 时即使已有 summary 也重新生成 |

**业务规则：**

1. 校验会议存在
2. `content` 为空 → 400
3. 调用 AI Service，解析结构化结果
4. 事务内更新三字段

**Response 200：**

```json
{
  "code": 0,
  "message": "AI 总结完成",
  "data": {
    "id": 1,
    "summary": "本次会议主要讨论...",
    "decisions": ["确定采用方案 A", "预算上调 10%"],
    "todos": [
      { "task": "输出 PRD", "completed": false },
      { "task": "安排技术评审", "completed": false }
    ],
    "updatedAt": "2026-07-27T06:30:00.000Z"
  }
}
```

**AI Prompt 输出建议（JSON）：**

```json
{
  "summary": "200字以内摘要",
  "decisions": ["决策1", "决策2"],
  "todos": [{ "task": "待办描述", "completed": false }]
}
```

---

### 3.7 PUT `/api/meetings/:id/todos/:index` — 更新待办状态

**URL：** `index` 为 `todos` 数组下标，从 **0** 开始。

**Request Body：**

```json
{ "completed": true }
```

或同时改文案：

```json
{ "completed": true, "task": "输出 PRD（已延期）" }
```

**Response 200：**

```json
{
  "code": 0,
  "message": "待办已更新",
  "data": {
    "index": 0,
    "todo": { "task": "输出 PRD", "completed": true },
    "todos": [
      { "task": "输出 PRD", "completed": true },
      { "task": "安排技术评审", "completed": false }
    ]
  }
}
```

**index 越界 → 404/400。**

## 四、后端分层结构建议

```
server/
├── app.js
├── config/
│   └── database.js
├── models/
│   ├── index.js
│   └── meeting.js
├── routes/
│   └── meetings.js
├── controllers/
│   └── meetingController.js
├── services/
│   ├── meetingService.js
│   └── aiService.js          # summarize 调用 LLM
├── middlewares/
│   ├── errorHandler.js
│   └── validate.js
└── utils/
    └── response.js           # 统一 { code, message, data }
```

**路由挂载：**

```javascript
// routes/meetings.js
const router = require("express").Router();
const ctrl = require("../controllers/meetingController");

router.post("/", ctrl.create);
router.get("/", ctrl.list);
router.get("/:id", ctrl.detail);
router.put("/:id", ctrl.update);
router.delete("/:id", ctrl.remove);
router.post("/:id/summarize", ctrl.summarize);
router.put("/:id/todos/:index", ctrl.updateTodo);

module.exports = router;

// app.js
app.use("/api/meetings", require("./routes/meetings"));
```

**注意：** 带 `:id` 的子路由若还有静态路径，应把 `/summarize` 写在通用 `/:id` 之前，或都用明确路径避免冲突（当前顺序已满足）。

## 六、Vue2 前端对接要点

1. **axios 封装**：`baseURL: '/api'`，拦截器统一处理 `code !== 0`。
2. **列表页**：`GET /meetings?page=&pageSize=&keyword=`。
3. **详情页**：进入后 `GET /meetings/:id`；有 `content` 无 `summary` 时展示「生成摘要」按钮 → `POST .../summarize`。
4. **待办勾选**：`PUT .../todos/:index`，本地可乐观更新，失败再回滚。
5. **字段映射**：后端 `underscored: true` 时，可在 axios 层做 camelCase ↔ snake_case，或前后端统一用一种命名。
